<script>
(function() {


    const root = document.querySelector('.premium-house-site');


    if (!root) return;




    /* ================= MOBILE MENU ================= */


    const menuBtn = root.querySelector('.phs-menu-btn');
    const mobileMenu = root.querySelector('.phs-mobile-menu');
    const menuClose = root.querySelector('.phs-menu-close');


    if (menuBtn && mobileMenu) {


        menuBtn.addEventListener('click', function() {
            mobileMenu.classList.add('open');
        });


    }


    if (menuClose && mobileMenu) {


        menuClose.addEventListener('click', function() {
            mobileMenu.classList.remove('open');
        });


    }


    mobileMenu.querySelectorAll('a').forEach(function(link) {


        link.addEventListener('click', function() {
            mobileMenu.classList.remove('open');
        });


    });




    /* ================= POPUP ================= */


    const popup = root.querySelector('#phs-popup');
    const popupClose = root.querySelector('.phs-popup-close');


    root.querySelectorAll('.phs-open-popup').forEach(function(button) {


        button.addEventListener('click', function() {


            if (popup) {
                popup.classList.add('open');
                document.body.style.overflow = 'hidden';
            }


        });


    });




    function closePopup() {


        if (popup) {
            popup.classList.remove('open');
            document.body.style.overflow = '';
        }


    }




    if (popupClose) {
        popupClose.addEventListener('click', closePopup);
    }




    if (popup) {


        popup.addEventListener('click', function(e) {


            if (e.target === popup) {
                closePopup();
            }


        });


    }




    /* ================= SMOOTH SCROLL ================= */


    root.querySelectorAll('a[href^="#"]').forEach(function(link) {


        link.addEventListener('click', function(e) {


            const id = this.getAttribute('href');


            if (!id || id === '#') return;


            const target = document.querySelector(id);


            if (target) {


                e.preventDefault();


                const header = root.querySelector('.phs-header');
                const offset = header ? header.offsetHeight : 0;


                const position =
                    target.getBoundingClientRect().top +
                    window.pageYOffset -
                    offset;


                window.scrollTo({
                    top:position,
                    behavior:'smooth'
                });


            }


        });


    });




    /* ================= FAQ ================= */


    root.querySelectorAll('.phs-faq-item').forEach(function(item) {


        const question = item.querySelector('.phs-faq-q');


        if (!question) return;


        question.addEventListener('click', function() {


            const active =
                item.classList.contains('active');


            root.querySelectorAll('.phs-faq-item').forEach(function(other) {
                other.classList.remove('active');
            });


            if (!active) {
                item.classList.add('active');
            }


        });


    });




    /* ================= CALCULATOR ================= */


    const areaInput =
        root.querySelector('#phs-calc-area');


    const materialSelect =
        root.querySelector('#phs-calc-material');


    const totalElement =
        root.querySelector('#phs-calc-total');


    function calculatePrice() {


        if (!areaInput || !materialSelect || !totalElement) {
            return;
        }


        let area =
            parseFloat(areaInput.value) || 0;


        if (area < 50) {
            area = 50;
        }


        const price =
            parseInt(materialSelect.value) || 350000;


        const total =
            area * price;


        totalElement.textContent =
            total.toLocaleString('ru-RU') + ' ₸';


    }




    if (areaInput) {
        areaInput.addEventListener('input', calculatePrice);
    }


    if (materialSelect) {
        materialSelect.addEventListener('change', calculatePrice);
    }


    const calcButton =
        root.querySelector('#phs-calc-button');


    if (calcButton) {
        calcButton.addEventListener('click', calculatePrice);
    }


    calculatePrice();




    /* ================= REVIEWS SLIDER ================= */


    const slider =
        root.querySelector('.phs-reviews-track');


    const slides =
        slider ? slider.children : [];


    const prev =
        root.querySelector('#phs-prev');


    const next =
        root.querySelector('#phs-next');


    let currentSlide = 0;




    function updateSlider() {


        if (!slider || !slides.length) return;


        slider.style.transform =
            'translateX(-' + (currentSlide * 100) + '%)';


    }




    if (next) {


        next.addEventListener('click', function() {


            currentSlide++;


            if (currentSlide >= slides.length) {
                currentSlide = 0;
            }


            updateSlider();


        });


    }




    if (prev) {


        prev.addEventListener('click', function() {


            currentSlide--;


            if (currentSlide < 0) {
                currentSlide = slides.length - 1;
            }


            updateSlider();


        });


    }




    /* ================= DEMO FORMS ================= */


    root.querySelectorAll('.phs-demo-form').forEach(function(form) {


        form.addEventListener('submit', function(e) {


            e.preventDefault();


            /*
             * Сейчас это демонстрационная форма.
             *
             * Позже сюда можно подключить:
             *
             * 1. Marquiz
             * 2. Яндекс.Метрику
             * 3. CRM
             * 4. Telegram
             * 5. n8n
             */


            const formWrap =
                form.closest('.phs-popup-form-wrap');


            if (formWrap) {


                formWrap.style.display = 'none';


                const success =
                    formWrap.parentElement.querySelector(
                        '.phs-popup-success'
                    );


                if (success) {
                    success.style.display = 'block';
                }


            } else {


                alert(
                    'Демонстрационная заявка отправлена. ' +
                    'На рабочем сайте здесь будет подключён Marquiz.'
                );


                form.reset();


            }


        });


    });




    /* ================= ANIMATIONS ================= */


    const observer =
        new IntersectionObserver(function(entries) {


            entries.forEach(function(entry) {


                if (entry.isIntersecting) {


                    entry.target.classList.add('visible');


                    observer.unobserve(entry.target);


                }


            });


        }, {
            threshold:.1
        });




    root.querySelectorAll('.phs-fade').forEach(function(element) {
        observer.observe(element);
    });




    /* ================= HEADER ================= */


    const header =
        root.querySelector('.phs-header');


    window.addEventListener('scroll', function() {


        if (!header) return;


        if (window.scrollY > 80) {


            header.style.background =
                'rgba(28,28,28,.96)';


            header.style.boxShadow =
                '0 4px 25px rgba(0,0,0,.15)';


        } else {


            header.style.background =
                'rgba(28,28,28,.25)';


            header.style.boxShadow =
                'none';


        }


    });




    /* ================= SCROLL TOP ================= */


    const scrollTop =
        root.querySelector('#phs-scroll-top');


    window.addEventListener('scroll', function() {


        if (!scrollTop) return;


        if (window.scrollY > 600) {
            scrollTop.classList.add('show');
        } else {
            scrollTop.classList.remove('show');
        }


    });




    if (scrollTop) {


        scrollTop.addEventListener('click', function() {


            window.scrollTo({
                top:0,
                behavior:'smooth'
            });


        });


    }


})();
</script>
